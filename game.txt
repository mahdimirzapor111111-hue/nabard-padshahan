    const SFX = {
        ctx: null,
        ensure() {
            if (!this.ctx) { const AC = window.AudioContext || window.webkitAudioContext; if (AC) this.ctx = new AC(); }
            if (this.ctx && this.ctx.state === 'suspended') this.ctx.resume();
            return this.ctx;
        },
        tone(f, d, type = 'sine', vol = 0.13, slide) {
            const c = this.ensure(); if (!c) return;
            const o = c.createOscillator(), g = c.createGain();
            o.type = type; o.frequency.setValueAtTime(f, c.currentTime);
            if (slide) o.frequency.exponentialRampToValueAtTime(Math.max(30, slide), c.currentTime + d);
            g.gain.setValueAtTime(vol, c.currentTime);
            g.gain.exponentialRampToValueAtTime(0.001, c.currentTime + d);
            o.connect(g).connect(c.destination); o.start(); o.stop(c.currentTime + d);
        },
        noise(d = 0.15, vol = 0.15) {
            const c = this.ensure(); if (!c) return;
            const b = c.createBuffer(1, c.sampleRate * d, c.sampleRate);
            const data = b.getChannelData(0);
            for (let i = 0; i < data.length; i++) data[i] = (Math.random() * 2 - 1) * (1 - i / data.length);
            const s = c.createBufferSource(); s.buffer = b;
            const g = c.createGain(); g.gain.value = vol;
            const f = c.createBiquadFilter(); f.type = 'lowpass'; f.frequency.value = 900;
            s.connect(f).connect(g).connect(c.destination); s.start();
        },
        play(name) {
            if (!soundOn) return;
            try {
                switch (name) {
                    case 'select':  this.tone(600, .07, 'square', .07); break;
                    case 'attack':  this.noise(.18, .16); this.tone(220, .2, 'sawtooth', .09, 80); break;

    const Music = {
        timer: null,
        bar: 0,
        master: null,
        customAudio: null,
        musicOn: loadLS('nabard_music_v1', true),
        volume: loadLS('nabard_music_vol_v1', 35),

        ensureMaster() {
            const c = SFX.ensure();
            if (!c) return null;
            if (!this.master) {
                this.master = c.createGain();
                this.master.connect(c.destination);
            }
            this.master.gain.value = this.volume / 100;
            return this.master;
        },
        isPlaying() {
            if (this.customAudio) return !this.customAudio.paused;
            return !!this.timer;
        },
        start() {
            if (!this.musicOn) return;
            if (this.customAudio) { this.customAudio.play().catch(()=>{}); return; }
            if (this.timer) return;
            if (!this.ensureMaster()) return;
            this.timer = setInterval(() => this.playBar(), 2400);
            this.playBar();
        },
        stop() {
            if (this.customAudio) this.customAudio.pause();
            clearInterval(this.timer);
            this.timer = null;
        },
        setVolume(v) {
            this.volume = v;
            saveLS('nabard_music_vol_v1', v);
            if (this.customAudio) this.customAudio.volume = v / 100;
            if (this.master) this.master.gain.value = v / 100;
        },
        useCustom(file) {
            try {
                if (this.customAudio) this.customAudio.pause();
                this.customAudio = new Audio(URL.createObjectURL(file));
                this.customAudio.loop = true;
                this.customAudio.volume = this.volume / 100;
                clearInterval(this.timer); this.timer = null;
                this.customAudio.play().catch(()=>{});
            } catch(e) { toast('⚠️ پخش این فایل ممکن نشد'); }
        },
        clearCustom() {
            if (this.customAudio) this.customAudio.pause();
            this.customAudio = null;
            this.start();
        },

        // --- سازهای موسیقی حماسی ---
        kick(t) {
            const c = SFX.ctx, out = this.master;
            const o = c.createOscillator(), g = c.createGain();
            o.type = 'sine';
            o.frequency.setValueAtTime(120, t);
            o.frequency.exponentialRampToValueAtTime(45, t + 0.12);
            g.gain.setValueAtTime(0.45, t);
            g.gain.exponentialRampToValueAtTime(0.001, t + 0.22);
            o.connect(g); g.connect(out);
            o.start(t); o.stop(t + 0.25);
        },
        snare(t) {
            const c = SFX.ctx, out = this.master;
            const b = c.createBuffer(1, Math.floor(c.sampleRate * 0.15), c.sampleRate);
            const d = b.getChannelData(0);
            for (let i = 0; i < d.length; i++) d[i] = (Math.random()*2-1) * Math.pow(1 - i/d.length, 2);
            const s = c.createBufferSource(); s.buffer = b;
            const f = c.createBiquadFilter(); f.type = 'highpass'; f.frequency.value = 1200;
            const g = c.createGain(); g.gain.value = 0.13;
            s.connect(f); f.connect(g); g.connect(out);
            s.start(t);
        },
        hat(t) {
            const c = SFX.ctx, out = this.master;
            const b = c.createBuffer(1, Math.floor(c.sampleRate * 0.05), c.sampleRate);
            const d = b.getChannelData(0);
            for (let i = 0; i < d.length; i++) d[i] = (Math.random()*2-1) * (1 - i/d.length);
            const s = c.createBufferSource(); s.buffer = b;
            const f = c.createBiquadFilter(); f.type = 'highpass'; f.frequency.value = 6000;
            const g = c.createGain(); g.gain.value = 0.05;
            s.connect(f); f.connect(g); g.connect(out);
            s.start(t);
        },
        bassNote(freq, t) {
            const c = SFX.ctx, out = this.master;
            const o = c.createOscillator(), g = c.createGain();
            o.type = 'triangle'; o.frequency.value = freq;
            g.gain.setValueAtTime(0.001, t);
            g.gain.linearRampToValueAtTime(0.2, t + 0.03);
            g.gain.exponentialRampToValueAtTime(0.001, t + 1.0);
            o.connect(g); g.connect(out);
            o.start(t); o.stop(t + 1.05);
        },
        pad(freqs, t, dur) {
            const c = SFX.ctx, out = this.master;
            freqs.forEach(f => {
                const o = c.createOscillator(), g = c.createGain();
                const flt = c.createBiquadFilter();
                o.type = 'sawtooth'; o.frequency.value = f;
                flt.type = 'lowpass'; flt.frequency.value = 850;
                g.gain.setValueAtTime(0.001, t);
                g.gain.linearRampToValueAtTime(0.045, t + 0.4);
                g.gain.linearRampToValueAtTime(0.001, t + dur);
                o.connect(flt); flt.connect(g); g.connect(out);
                o.start(t); o.stop(t + dur + 0.05);
            });
        },
        lead(freq, t) {
            const c = SFX.ctx, out = this.master;
            const o = c.createOscillator(), g = c.createGain();
            o.type = 'square'; o.frequency.value = freq;
            g.gain.setValueAtTime(0.001, t);
            g.gain.linearRampToValueAtTime(0.06, t + 0.02);
            g.gain.exponentialRampToValueAtTime(0.001, t + 0.5);
            o.connect(g); g.connect(out);
            o.start(t); o.stop(t + 0.55);
        },
        playBar() {
            const c = SFX.ctx;
            if (!c || !this.master) return;
            const t = c.currentTime + 0.05;
            const bar = this.bar % 8;
            // پیشروی آکورد: Am - F - G - Em (حس حماسی)
            const prog = [
                [110.0, 130.8, 164.8],
                [87.3, 110.0, 130.8],
                [98.0, 123.5, 146.8],
                [82.4, 98.0, 123.5],
            ];
            const chord = prog[Math.floor(bar / 2)];
            // درام‌ها
            this.kick(t); this.kick(t + 0.6);
            this.snare(t + 1.2); this.snare(t + 1.8);
            if (bar % 2 === 1) { this.hat(t + 0.3); this.hat(t + 0.9); this.hat(t + 1.5); }
            // بیس
            this.bassNote(chord[0] / 2, t);
            this.bassNote(chord[0] / 2, t + 1.2);
            // پد آکوردی
            this.pad(chord, t, 2.3);
            // ملودی در نیمه دوم
            if (bar >= 4) {
                const scale = [440, 523.3, 587.3, 659.3, 784];
                const idx = (this.bar * 3) % scale.length;
                this.lead(scale[idx], t + 0.6);
                this.lead(scale[(idx + 2) % scale.length], t + 1.5);
            }
            this.bar++;
        }
    };

    function buildGameBoard() {
        board = Array.from({length:6}, () => Array(3).fill(null));
        ROW_IDS.forEach(id => $(id).innerHTML = '');
        ensureSidesPopulated();
        for (let r = 0; r < 6; r++) for (let c = 0; c < 3; c++) {
            const e = editorLayout[r][c];
            if (!e) continue;
            const isP = r >= 3;
            let el;
            if (e.kind === 'default') {
                const d = CARD_DATA[e.id];
                el = makeCardEl({...d, id: e.id}, isP, false);
            } else if (e.kind === 'draft') {
                el = makeCardEl(deepCopy(editorCard), isP, true);
            } else {
                el = makeCardEl(deepCopy(e.data), isP, true);
            }
            el.addEventListener('click', (ev) => onCardClick(ev, el));
            putInSlot(r, c, el);
        }
        layoutDirty = false;
    }

    function sideHasOtherAlive(el) {
        const isP = el.dataset.isPlayer === 'true';
        const r0 = isP ? 3 : 0;
        let n = 0;
        for (let r = r0; r < r0+3; r++) for (let c = 0; c < 3; c++) {
            const o = cardAt(r, c);
            if (o && o !== el) n++;
        }
        return n > 0;
    }
    function tauntBlocks(target, attacker) {
        if (attacker && attacker.dataset.ignoreGuard === '1') return false;
        const isPT = target.dataset.isPlayer === 'true';
        const r0 = isPT ? 3 : 0;
        const taunts = [];
        for (let r = r0; r < r0+3; r++) for (let c = 0; c < 3; c++) {
            const el = cardAt(r, c);
            if (el && el.dataset.taunt === '1') taunts.push(el);
        }
        if (!taunts.length || target.dataset.taunt === '1') return false;
        return taunts.some(t => {
            const r = +t.parentElement.dataset.row, c = +t.parentElement.dataset.col;
            const g = protectedBy(r, c);
            return !(g && (g.dataset.type === 'defender' || g.dataset.type === 'both'));
        });
    }
    function onCardClick(e, el) {
        e.stopPropagation();
        if (inspectMode) { showCardInfo(el); return; }
        if (currentTurn !== 'player' || busy || gameOver) return;
        const isP = el.dataset.isPlayer === 'true';
        if (isP) {
            if (el.dataset.dead === '1') return;
            if (el.dataset.type === 'spell') { showFloat(el, '📜 افسون قابل انتخاب نیست', '#ffd54f'); return; }
            if (el.dataset.frozen === '1') { showFloat(el, '❄️ یخ زده!', '#4fc3f7'); return; }
            if (el.dataset.hasAttacked === 'true') { showFloat(el, '⏳ قبلاً حمله کرده', '#ff5252'); clearSelection(); return; }
            if (selectedCard === el) { clearSelection(); return; }
            clearSelection();
            selectedCard = el; el.classList.add('selected');
            SFX.play('select');
            return;
        }
        if (el.dataset.dead === '1') return;
        if (!selectedCard) return;
        const attacker = selectedCard;
        if (el.dataset.stealth === '1' && sideHasOtherAlive(el)) {
            showFloat(el, '👻 قابل هدف نیست!', '#90a4ae'); return;
        }
        if (attacker.dataset.ignoreGuard !== '1') {
            const tR = +el.parentElement.dataset.row, tC = +el.parentElement.dataset.col;
            const guardRow = tR === 0 ? 1 : tR === 1 ? 2 : -1;
            if (guardRow !== -1) {
                const g = cardAt(guardRow, tC);
                if (g && (g.dataset.type === 'defender' || g.dataset.type === 'both')) {
                    showFloat(el, '🛡️ محفوظ!', '#2196f3'); SFX.play('thorns'); clearSelection(); return;
                }
            }
            if (tauntBlocks(el, attacker)) { showFloat(el, '🎯 اول کارت تحریک‌شده!', '#ff8a80'); SFX.play('taunt'); return; }
        }
        tryPlayerAttack(attacker, el);
    }

    async function tryPlayerAttack(attacker, target) {

    async function performAttack(attacker, target) {
        const s = battleSeq;
        attacker.dataset.hasAttacked = 'true';
        attacker.classList.remove('can-attack', 'selected');
        const aR = attacker.getBoundingClientRect(), tR = target.getBoundingClientRect();
        const dx = tR.left - aR.left, dy = tR.top - aR.top;
        const up = attacker.dataset.isPlayer === 'true';
        attacker.classList.add('attacking'); attacker.style.zIndex = '999';
        SFX.play('attack');
        attacker.style.transition = 'transform .26s ease-out';
        attacker.style.transform = `translate(${dx*0.4}px,${dy*0.4}px) scale(1.15) rotate(${up?'-8':'8'}deg)`;
        await sleep(260);
        if (s !== battleSeq) { resetAnim(attacker); return; }
        attacker.style.transition = 'transform .3s ease-in';
        attacker.style.transform = `translate(${dx}px,${dy}px) scale(1.25) rotate(${up?'12':'-12'}deg)`;
        await sleep(300);
        if (s !== battleSeq) { resetAnim(attacker); return; }

        const targetSideP = target.dataset.isPlayer === 'true';

        // 💨 جاخالی
        const dodge = +target.dataset.dodge || 0;
        if (dodge > 0 && Math.random() * 100 < dodge) {
            showFloat(target, '💨 جاخالی!', '#b0bec5');
            SFX.play('dodge');
            logAdd(`💨 ${target.dataset.name} حمله ${attacker.dataset.name} را جاخالی داد`);
            await sleep(300);
        } else {
            // 💥 بحرانی
            let dmg = parseInt(attacker.dataset.attack) || 0;
            const critP = +attacker.dataset.crit || 0;
            if (critP > 0 && Math.random() * 100 < critP) {
                dmg *= 2;
                showFloat(attacker, '💥 بحرانی!', '#ffd54f');
                SFX.play('crit');
                logAdd(`💥 ${attacker.dataset.name} ضربه بحرانی زد!`);
                await sleep(250);
                if (s !== battleSeq) { resetAnim(attacker); return; }
            }
            // 🔊 صدای حمله شخصی + افکت برش و فلاش
            playCardSound(attacker, 'attack');
            attackImpactFX(target, dmg);
            const dealt = dealDamage(target, dmg, attacker);
            logAdd(`⚔️ ${attacker.dataset.name} به ${target.dataset.name} ${dealt} آسیب زد`);
            if (attacker.dataset.isPlayer === 'true' && dealt > 0) { stats.damage += dealt; saveLS('nabard_stats_v2', stats); }
            await sleep(120);

            // 🪓 برش جانبی
            if (attacker.dataset.cleave === '1' && s === battleSeq) {
                const tr = +target.parentElement.dataset.row, tc = +target.parentElement.dataset.col;
                [tc-1, tc+1].forEach(nc => {
                    if (nc < 0 || nc > 2) return;
                    const nb = cardAt(tr, nc);
                    if (nb) {
                        const half = Math.floor(dmg / 2);
                        if (half > 0) { showFloat(nb, '🪓', '#ff7043'); dealDamage(nb, half, attacker); }
                    }
                });
            }

            // 🛡️ فاز دفاع هدف
            if (isAlive(target)) await runPhase(target, 'onDefend', { attacker }, s);

            // 🌵 خار پسیو هدف
            if (isAlive(target)) {
                const th = +target.dataset.thorns || 0;
                if (th > 0) {
                    showFloat(target, '🌵 پاتک', '#4fc3f7');
                    SFX.play('thorns');
                    await sleep(200);
                    if (s !== battleSeq) { resetAnim(attacker); return; }
                    dealDamage(attacker, th, target);
                    logAdd(`🌵 خارِ ${target.dataset.name}: ${th} آسیب به ${attacker.dataset.name}`);
                }
            }
            // فاز حمله مهاجم
            if (isAlive(attacker)) await runPhase(attacker, 'onAttack', null, s);

            // 🗡️ حمله دوگانه
            if (isAlive(attacker) && attacker.dataset.doubleStrike === '1' && s === battleSeq) {
                await sleep(250);
                if (s !== battleSeq) { resetAnim(attacker); return; }
                let t2 = isAlive(target) ? target : randomEnemyOfSide(targetSideP);
                if (t2) {
                    showFloat(attacker, '🗡️ ضربه دوم!', '#ff8a80');
                    await sleep(200);
                    if (s !== battleSeq) { resetAnim(attacker); return; }
                    playCardSound(attacker, 'attack');
                    attackImpactFX(t2, parseInt(attacker.dataset.attack) || 0);
                    const d2 = dealDamage(t2, parseInt(attacker.dataset.attack) || 0, attacker);
                    logAdd(`🗡️ حمله دوگانه ${attacker.dataset.name}: ${d2} آسیب به ${t2.dataset.name}`);
                    if (isAlive(t2)) await runPhase(t2, 'onDefend', { attacker }, s);
                }
            }
        }
        if (s !== battleSeq) { resetAnim(attacker); return; }
        await flushOnKills(s);
        if (s !== battleSeq) { resetAnim(attacker); return; }

        // برگشت
        if (isAlive(attacker)) {
            attacker.style.transition = 'transform .35s ease-in-out';
            attacker.style.transform = 'translate(0,0) scale(1) rotate(0)';
            await sleep(380);
        }
        resetAnim(attacker);
    }
    function resetAnim(el) {
        if (!el || !el.isConnected) return;
        el.classList.remove('attacking');
        el.style.zIndex = ''; el.style.transition = ''; el.style.transform = '';
    }
    function randomEnemyOfSide(sideIsPlayer) {
        const list = [];
        for (let r = sideIsPlayer?3:0; r < (sideIsPlayer?6:3); r++)
            for (let c = 0; c < 3; c++) { const el = cardAt(r,c); if (el) list.push(el); }
        return list.length ? list[Math.floor(Math.random()*list.length)] : null;
    }

    // ================= آسیب / مرگ =================
    function dealDamage(target, amount, source) {
        if (!isAlive(target)) return 0;
        let dmg = Math.max(0, parseInt(amount) || 0);
        if (dmg <= 0) return 0;
        const pierce = source && source.dataset && source.dataset.pierce === '1';
        // 🛡️ سپر
        let sh = +target.dataset.shield || 0;
        if (sh > 0 && !pierce) {
            const absorbed = Math.min(sh, dmg);
            sh -= absorbed; dmg -= absorbed;
            target.dataset.shield = String(sh);
            showFloat(target, `🛡️ ${absorbed} جذب شد`, '#4fc3f7');
            flashCard(target, 'shield-flash');
            SFX.play('shield');
            updateStatus(target);
            if (dmg <= 0) return absorbed;
        } else if (sh > 0 && pierce) {
            showFloat(target, '⚡ سپر شکست!', '#f9a825');
        }
        let hp = (parseInt(target.dataset.health) || 0) - dmg;
        // 💪 جان سخت
        if (hp <= 0 && target.dataset.lastStand === '1' && target.dataset.lastStandUsed !== '1') {
            target.dataset.lastStandUsed = '1';
            hp = 1;
            showFloat(target, '💪 جان سخت!', '#4db6ac');
            SFX.play('revive');
            logAdd(`💪 ${target.dataset.name} با جان سخت زنده ماند!`);
            updateStatus(target);
        }
        // 👼 رستاخیز
        if (hp <= 0 && +target.dataset.resurrect > 0 && target.dataset.resurrectUsed !== '1') {
            target.dataset.resurrectUsed = '1';
            const rsP = +target.dataset.resurrect;
            hp = Math.max(1, Math.ceil((+target.dataset.maxHealth) * rsP / 100));
            showFloat(target, '👼 رستاخیز!', '#ce93d8');
            SFX.play('revive');
            target.classList.add('revive-glow');
            setTimeout(() => target.classList.remove('revive-glow'), 700);
            logAdd(`👼 ${target.dataset.name} با رستاخیز زنده شد (${hp} جون)!`);
            updateStatus(target);
        }
        target.dataset.health = String(hp);
        updateHP(target);
        target.classList.add('hit');
        setTimeout(() => target.classList.remove('hit'), 400);
        spawnParticles(target);
        showFloat(target, '-' + dmg, '#ff1744');
        SFX.play('hit');
        // 😤 خشم
        const bk = +target.dataset.berserk || 0;
        if (bk > 0 && hp > 0) {
            target.dataset.attack = String((+target.dataset.attack) + bk);
            updateAtk(target);
            showFloat(target, `😤 +${bk} ⚔️`, '#ff8a80');
        }
        if (hp <= 0) killCard(target, source);
        return dmg;
    }
    function killCard(card, killer) {
        if (!isAlive(card)) return;
        card.dataset.dead = '1';
        const r = +card.parentElement.dataset.row, c = +card.parentElement.dataset.col;
        if (board[r]) board[r][c] = null;
        card.classList.add('dying');
        // 🔊 صدای مرگ شخصی یا پیش‌فرض
        if (card.dataset.soundDeath) playCardSound(card, 'death');
        else SFX.play('death');
        // 💥 افکت مرگ سینمایی
        deathEffectFX(card);
        logAdd(`💀 ${card.dataset.name} نابود شد`);
        setTimeout(() => { if (card.isConnected) card.remove(); }, 560);
        if (isAlive(killer)) pendingOnKills.push({ killer, victim: card });
        if (card.dataset.type !== 'spell') triggerDeath(card, killer);
        scheduleWinCheck();
    }
    async function flushOnKills(s) {
        let guard = 0;
        while (pendingOnKills.length && guard < 10) {
            guard++;
            const k = pendingOnKills.shift();
            if (s !== battleSeq) { pendingOnKills = []; return; }
            if (!isAlive(k.killer)) continue;
            logAdd(`🎯 ${k.killer.dataset.name} کارت کشت!`);
            await runPhase(k.killer, 'onKill', { victim: k.victim }, s);
        }
        if (guard >= 10) pendingOnKills = [];
    }
    async function triggerDeath(card, killer) {
        const s = battleSeq;
        if (card.dataset.silenced === '1') {
            logAdd(`🔇 ${card.dataset.name} ساکت بود؛ قابلیت مرگ اجرا نشد`);
            return;
        }
        let ab; try { ab = JSON.parse(card.dataset.abilities || '{}'); } catch(e) { return; }
        for (const a of (ab.onDeath || [])) {
            if (s !== battleSeq) return;
            await sleep(300);
            if (s !== battleSeq) return;
            logAdd(`☠️ ${card.dataset.name} (بعد از مرگ): ${a.icon} ${a.label}`);
            await runAbility(card, a, { killer }, s);
        }
        if (s === battleSeq) await flushOnKills(s);
    }

    // ================= اجرای قابلیت‌ها =================
    async function runPhase(card, phase, ctx, seq) {
        const s = seq ?? battleSeq;
        if (card.dataset.silenced === '1') {
            showFloat(card, '🔇 ساکت!', '#90a4ae');
            return;
        }
        let ab; try { ab = JSON.parse(card.dataset.abilities || '{}'); } catch(e) { return; }
        for (const a of (ab[phase] || [])) {
            if (s !== battleSeq) return;
            if (!isAlive(card)) return;
            flashCard(card); SFX.play('ability');
            logAdd(`✨ ${card.dataset.name}: ${a.icon} ${a.label}`);
            await runAbility(card, a, ctx || {}, s);
            await sleep(350);
        }
        if (s === battleSeq) await flushOnKills(s);
    }
    function getAbilityTargets(source, ab) {
        const srcP = source.dataset.isPlayer === 'true';
        const want = (ab.targetSide === 'ally') ? srcP : !srcP;
        const out = [];
        (ab.positions || []).forEach(([r, c]) => {
            const el = cardAt(r, c);
            if (el && el.dataset.isPlayer === String(want)) out.push(el);
        });
        return out;
    }
    function summonMinions(source, stat) {
        const side = source.dataset.isPlayer === 'true';
        const r0 = side ? 3 : 0;
        const sRow = +source.parentElement.dataset.row;
        const empties = [];
        for (let c = 0; c < 3; c++) for (let r = r0; r < r0+3; r++) if (!cardAt(r, c)) empties.push([r, c]);
        empties.sort((a, b) => Math.abs(a[0]-sRow) - Math.abs(b[0]-sRow));
        if (!empties.length) { showFloat(source, '⛔ جا خالی نیست', '#ff8a80'); return; }
        const [r, c] = empties[0];
        const def = { name: 'روح', type: 'attacker', tier: 'normal',
                      attack: clamp(stat, 0, 99), health: clamp(stat, 1, 99),
                      icon: '👻', image: null, borderColor: '#b0bec5',
                      sounds: defaultSounds(),
                      passives: defaultPassives(),
                      abilities: { onStart:[],onTurnStart:[],onAttack:[],onDefend:[],onKill:[],onDeath:[] } };
        const el = makeCardEl(def, side, true);
        el.addEventListener('click', ev => onCardClick(ev, el));
        if (side && currentTurn === 'player' && !busy) el.classList.add('can-attack');
        else { el.dataset.hasAttacked = 'true'; }
        putInSlot(r, c, el);
        showFloat(el, '👻 احضار شد!', '#ce93d8');
        SFX.play('summon');
        logAdd(`👻 روحی با ${stat} قدرت/جون احضار شد`);
    }
    async function runAbility(source, ab, ctx, seq) {
        const s = seq ?? battleSeq;
        const amt = Math.max(0, parseInt(ab.amount) || 0);
        const godBlock = (t) => { showFloat(t, '✨ مصون!', '#ffd54f'); };

        if (ab.action === 'thorns' && ctx && ctx.killer) {
            const k = ctx.killer;
            if (isAlive(k)) {
                showFloat(isAlive(source) ? source : k, '🌵 پاتک!', '#4fc3f7');
                SFX.play('thorns');
                dealDamage(k, amt, source);
                logAdd(`🌵 پاتکِ بعد از مرگ ${source.dataset.name}: ${amt} به ${k.dataset.name}`);
            }
            return;
        }

        switch (ab.action) {
            case 'counterAttack': {
                const t = ctx && ctx.attacker;
                if (isAlive(t)) {
                    showFloat(source, '🗡️ حمله متقابل!', '#ff8a80');
                    SFX.play('thorns');
                    dealDamage(t, amt, source);
                    logAdd(`🗡️ ${source.dataset.name} به حمله‌کننده ${amt} آسیب متقابل زد`);
                }
                break;
            }
            case 'lifesteal': {
                if (isAlive(source) && amt > 0) {
                    const mx = +source.dataset.maxHealth;
                    const before = +source.dataset.health;
                    const hp = Math.min(mx, before + amt);
                    source.dataset.health = String(hp);
                    updateHP(source);
                    showFloat(source, `+${hp - before} 🩸`, '#f06292');
                    flashCard(source, 'heal-flash');
                    SFX.play('heal');
                }
                break;
            }
            case 'execute': {
                for (const t of getAbilityTargets(source, ab)) {
                    if (s !== battleSeq) return;
                    if (!isAlive(t)) continue;
                    if (t.dataset.type === 'god') { godBlock(t); continue; }
                    if ((+t.dataset.health) <= amt) {
                        showFloat(t, '🪦 اعدام!', '#ff1744');
                        killCard(t, source);
                    } else {
                        showFloat(t, '🪦 جانش بیشتر است', '#90a4ae');
                    }
                    await sleep(300);
                }
                break;
            }
            case 'silence': {
                let n = 0;
                for (const t of getAbilityTargets(source, ab)) {
                    if (n >= amt) break;
                    if (!isAlive(t)) continue;
                    if (t.dataset.type === 'god') { godBlock(t); continue; }
                    t.dataset.silenced = '1';
                    updateStatus(t);
                    showFloat(t, '🔇 ساکت شد!', '#90a4ae');
                    n++;
                }
                if (n) { SFX.play('freeze'); logAdd(`🔇 ${n} کارت ساکت شدند (قابلیت‌هایشان قطع شد)`); }
                break;
            }
            case 'burn': {
                let any = false;
                getAbilityTargets(source, ab).forEach(t => {
                    if (!isAlive(t)) return;
                    if (t.dataset.type === 'god') { godBlock(t); return; }
                    const cur = +t.dataset.burn || 0;
                    t.dataset.burn = String(cur + amt);
                    updateStatus(t);
                    showFloat(t, `🔥+${amt}`, '#ff7043');
                    SFX.play('poison');
                    any = true;
                });
                if (any) logAdd(`🔥 آتش (${amt}) اعمال شد`);
                break;
            }
            case 'maxHpUp': {
                getAbilityTargets(source, ab).forEach(t => {
                    if (!isAlive(t)) return;
                    const mx = clamp((+t.dataset.maxHealth) + amt, 1, 99);
                    t.dataset.maxHealth = String(mx);
                    t.dataset.health = String(Math.min(mx, (+t.dataset.health) + amt));
                    updateHP(t);
                    showFloat(t, `💖+${amt}`, '#f06292');
                    flashCard(t, 'heal-flash');
                });
                SFX.play('heal');
                break;
            }
            case 'maxHpDown': {
                getAbilityTargets(source, ab).forEach(t => {
                    if (!isAlive(t)) return;
                    if (t.dataset.type === 'god') { godBlock(t); return; }
                    const mx = Math.max(1, (+t.dataset.maxHealth) - amt);
                    t.dataset.maxHealth = String(mx);
                    if ((+t.dataset.health) > mx) { t.dataset.health = String(mx); }
                    updateHP(t);
                    showFloat(t, `🩹-${amt}`, '#ff8a80');
                });
                SFX.play('hit');
                break;
            }
            case 'stealAttack': {
                let total = 0;
                getAbilityTargets(source, ab).forEach(t => {
                    if (!isAlive(t) || !isAlive(source)) return;
                    if (t.dataset.type === 'god') { godBlock(t); return; }
                    const take = Math.min(amt, +t.dataset.attack);
                    if (take <= 0) return;
                    t.dataset.attack = String((+t.dataset.attack) - take);
                    updateAtk(t);
                    total += take;
                    showFloat(t, `🥷-${take} ⚔️`, '#ce93d8');
                });
                if (total > 0 && isAlive(source)) {
                    source.dataset.attack = String((+source.dataset.attack) + total);
                    updateAtk(source);
                    showFloat(source, `🥷+${total} ⚔️`, '#ffd54f');
                    SFX.play('steal');
                    logAdd(`🥷 ${source.dataset.name} ${total} قدرت دزدید`);
                }
                break;
            }
            case 'drain': {
                let total = 0;
                for (const t of getAbilityTargets(source, ab)) {
                    if (s !== battleSeq) return;
                    if (!isAlive(t)) continue;
                    total += dealDamage(t, amt, source);
                    await sleep(200);
                }
                if (total > 0 && isAlive(source)) {
                    const mx = +source.dataset.maxHealth;
                    const before = +source.dataset.health;
                    const hp = Math.min(mx, before + total);
                    source.dataset.health = String(hp);
                    updateHP(source);
                    showFloat(source, `🧛+${hp - before}`, '#f06292');
                    flashCard(source, 'heal-flash');
                    SFX.play('heal');
                    logAdd(`🧛 ${source.dataset.name} ${total} آسیب زد و ${hp - before} مکید`);
                }
                break;
            }
            case 'splash': {
                for (const t of getAbilityTargets(source, ab)) {
                    if (s !== battleSeq) return;
                    if (!isAlive(t)) continue;
                    const tr = +t.parentElement.dataset.row, tc = +t.parentElement.dataset.col;
                    showFloat(t, '🌊', '#4fc3f7');
                    dealDamage(t, amt, source);
                    [tc-1, tc+1].forEach(nc => {
                        if (nc < 0 || nc > 2) return;
                        const nb = cardAt(tr, nc);
                        if (nb) { const half = Math.floor(amt/2); if (half > 0) dealDamage(nb, half, source); }
                    });
                    await sleep(250);
                }
                break;
            }
            case 'cleanse': {
                getAbilityTargets(source, ab).forEach(t => {
                    if (!isAlive(t)) return;
                    t.dataset.poison = '0'; t.dataset.burn = '0';
                    t.dataset.frozen = '0'; t.dataset.silenced = '0';
                    t.classList.remove('frozen');
                    updateStatus(t);
                    showFloat(t, '🧼 پاک شد!', '#69f0ae');
                });
                SFX.play('heal');
                break;
            }
            case 'dispel': {
                getAbilityTargets(source, ab).forEach(t => {
                    if (!isAlive(t)) return;
                    if (t.dataset.type === 'god') { godBlock(t); return; }
                    t.dataset.shield = '0'; t.dataset.thorns = '0';
                    t.dataset.taunt = '0'; t.dataset.dodge = '0';
                    updateStatus(t);
                    showFloat(t, '🌪 جادو زدوده شد!', '#b0bec5');
                });
                SFX.play('dodge');
                break;
            }
            case 'summon': {
                summonMinions(source, amt || 1);
                break;
            }
            case 'healFull': {
                getAbilityTargets(source, ab).forEach(t => {
                    if (!isAlive(t)) return;
                    const before = +t.dataset.health;
                    t.dataset.health = t.dataset.maxHealth;
                    updateHP(t);
                    showFloat(t, `✨+${(+t.dataset.maxHealth) - before}`, '#69f0ae');
                    flashCard(t, 'heal-flash');
                });
                SFX.play('heal');
                break;
            }
            case 'destroyMage':     await destroyByType(getAbilityTargets(source, ab), 'mage', amt, s, source); break;
            case 'destroyDefender': await destroyByType(getAbilityTargets(source, ab), 'defender', amt, s, source); break;
            case 'destroyAttacker': await destroyByType(getAbilityTargets(source, ab), 'attacker', amt, s, source); break;
            case 'heal':            healCards(getAbilityTargets(source, ab), amt); break;
            case 'damage': {
                for (const t of getAbilityTargets(source, ab)) {
                    if (s !== battleSeq) return;
                    dealDamage(t, amt, source);
                    await sleep(200);
                }
                break;
            }
            case 'buffAttack': changeAttack(getAbilityTargets(source, ab), amt, true); break;
            case 'nerfAttack': changeAttack(getAbilityTargets(source, ab), amt, false); break;
            case 'freeze':     freezeCards(getAbilityTargets(source, ab), amt); break;
            case 'shield': {
                let any = false;
                getAbilityTargets(source, ab).forEach(t => {
                    if (!isAlive(t)) return;
                    const cur = +t.dataset.shield || 0;
                    t.dataset.shield = String(cur + amt);
                    updateStatus(t);
                    showFloat(t, `🛡️+${amt}`, '#4fc3f7');
                    flashCard(t, 'shield-flash');
                    any = true;
                });
                if (any) { SFX.play('shield'); logAdd(`🛡️ سپر ${amt} اعطا شد`); }
                break;
            }
            case 'poison': {
                let any = false;
                getAbilityTargets(source, ab).forEach(t => {
                    if (!isAlive(t)) return;
                    if (t.dataset.type === 'god') { godBlock(t); return; }
                    const cur = +t.dataset.poison || 0;
                    t.dataset.poison = String(cur + amt);
                    updateStatus(t);
                    showFloat(t, `☠️+${amt}`, '#ce93d8');
                    SFX.play('poison');
                    any = true;
                });
                if (any) logAdd(`☠️ زهر (${amt}) اعمال شد`);
                break;
            }
            case 'taunt': {
                let any = false;
                getAbilityTargets(source, ab).forEach(t => {
                    if (!isAlive(t)) return;
                    t.dataset.taunt = '1';
                    updateStatus(t);
                    showFloat(t, '🎯 تحریک!', '#ff8a80');
                    any = true;
                });
                if (any) { SFX.play('taunt'); logAdd('🎯 کارت تحریک شد — باید اول به آن حمله شود'); }
                break;
            }
            case 'dodge': {
                let any = false;
                getAbilityTargets(source, ab).forEach(t => {
                    if (!isAlive(t)) return;
                    const d = Math.min(90, (+t.dataset.dodge || 0) + amt);
                    t.dataset.dodge = String(d);
                    updateStatus(t);
                    showFloat(t, `💨${d}%`, '#b0bec5');
                    any = true;
                });
                if (any) SFX.play('dodge');
                break;
            }
            case 'thorns': {
                getAbilityTargets(source, ab).forEach(t => {
                    if (!isAlive(t)) return;
                    const th = (+t.dataset.thorns || 0) + amt;
                    t.dataset.thorns = String(th);
                    updateStatus(t);
                    showFloat(t, '🌵 خار ' + th, '#4fc3f7');
                });
                SFX.play('buff');
                break;
            }
        }
    }
    async function destroyByType(targets, type, count, s, source) {
        let killed = 0;
        for (const t of targets) {
            if (killed >= count) break;
            if (s !== battleSeq) return;
            if (!isAlive(t)) continue;
            if (t.dataset.type === 'god') { showFloat(t, '✨ مصون!', '#ffd54f'); continue; }
            showFloat(t, '💀 نابودی!', '#ff1744');
            killCard(t, source);
            killed++;
            await sleep(300);
        }
    }
    function healCards(targets, amount) {
        let any = false;
        targets.forEach(t => {
            if (!isAlive(t)) return;
            const mx = +t.dataset.maxHealth;
            const before = +t.dataset.health;
            const hp = Math.min(mx, before + amount);
            if (hp === before) { showFloat(t, 'سلامت کامل', '#9e9e9e'); return; }
            t.dataset.health = String(hp);
            updateHP(t);
            showFloat(t, '+' + (hp - before) + ' ❤️', '#69f0ae');
            flashCard(t, 'heal-flash');
            any = true;
        });
        if (any) SFX.play('heal');
    }
    function changeAttack(targets, amount, isBuff) {
        if (amount <= 0) return;
        let any = false;
        targets.forEach(t => {
            if (!isAlive(t)) return;
            if (!isBuff && t.dataset.type === 'god') { showFloat(t, '✨ مصون!', '#ffd54f'); return; }
            let a = (+t.dataset.attack) + (isBuff ? amount : -amount);
            if (a < 0) a = 0;
            t.dataset.attack = String(a);
            updateAtk(t);
            showFloat(t, (isBuff ? '+' : '-') + amount + ' ⚔️', isBuff ? '#ffd54f' : '#ff8a80');
            any = true;
        });
        if (any) SFX.play(isBuff ? 'buff' : 'hit');
    }
    function freezeCards(targets, count) {
        let n = 0;
        for (const t of targets) {
            if (n >= count) break;
            if (!isAlive(t)) continue;
            if (t.dataset.type === 'god') { showFloat(t, '✨ مصون!', '#ffd54f'); continue; }
            t.dataset.frozen = '1';
            t.classList.add('frozen');
            t.classList.remove('can-attack', 'selected');
            if (selectedCard === t) selectedCard = null;
            showFloat(t, '❄️ یخ زد!', '#4fc3f7');
            n++;
        }
        if (n) { SFX.play('freeze'); logAdd(`❄️ ${n} کارت یخ زدند`); }
    }

    async function runStartPhase(s) {
        const cards = [];
        document.querySelectorAll('#game-board .card').forEach(el => {
            if (isAlive(el)) cards.push(el);
        });
        for (const el of cards) {
            if (s !== battleSeq) return;
            // 🔊 صدای شروع شخصی
            if (el.dataset.soundStart) {
                playCardSound(el, 'start');
                await sleep(250);
                if (s !== battleSeq) return;
            }
            await runPhase(el, 'onStart', null, s);
            if (s !== battleSeq) return;
            if (el.dataset.type === 'spell' && isAlive(el)) {
                showFloat(el, '✨ افسون!', '#ffd54f');
                await sleep(350);
                if (s !== battleSeq) return;
                el.dataset.dead = '1';
                const r = +el.parentElement.dataset.row, c = +el.parentElement.dataset.col;
                if (board[r]) board[r][c] = null;
                el.classList.add('dying');
                setTimeout(() => el.remove(), 500);
                logAdd(`📜 افسون «${el.dataset.name}» مصرف شد`);
                await sleep(200);
            }
        }
    }
    async function tickDoTSide(isP, s) {
        const cards = [];
        document.querySelectorAll('#game-board .card').forEach(el => {
            if (el.dataset.isPlayer === String(isP) && isAlive(el) &&
                ((+el.dataset.poison || 0) > 0 || (+el.dataset.burn || 0) > 0)) cards.push(el);
        });
        for (const el of cards) {
            if (s !== battleSeq || gameOver) return;
            const po = +el.dataset.poison || 0;
            const bu = +el.dataset.burn || 0;
            if (po > 0) {
                showFloat(el, '☠️ زهر ' + po, '#ce93d8');
                SFX.play('poison');
                dealDamage(el, po, null);
                logAdd(`☠️ ${el.dataset.name} ${po} آسیب زهر خورد`);
                if (isAlive(el)) { el.dataset.poison = String(Math.max(0, po - 1)); updateStatus(el); }
                await sleep(350);
            }
            if (s !== battleSeq || gameOver) return;
            if (bu > 0 && isAlive(el)) {
                showFloat(el, '🔥 آتش ' + bu, '#ff7043');
                SFX.play('poison');
                dealDamage(el, bu, null);
                logAdd(`🔥 ${el.dataset.name} ${bu} آسیب آتش خورد`);
                if (isAlive(el)) { el.dataset.burn = String(Math.max(0, bu - 1)); updateStatus(el); }
                await sleep(350);
            }
        }
    }
    async function runSidePassives(isP, s) {
        const cards = [];
        document.querySelectorAll('#game-board .card').forEach(el => {
            if (el.dataset.isPlayer === String(isP) && isAlive(el)) cards.push(el);
        });
        for (const el of cards) {
            if (s !== battleSeq || gameOver) return;
            const rg = +el.dataset.regen || 0;
            if (rg > 0) {
                const mx = +el.dataset.maxHealth;
                const before = +el.dataset.health;
                const hp = Math.min(mx, before + rg);
                if (hp > before) {
                    el.dataset.health = String(hp);
                    updateHP(el);
                    showFloat(el, `+${hp - before} 💚`, '#69f0ae');
                    SFX.play('heal');
                }
            }
            const ra = +el.dataset.rage || 0;
            if (ra > 0) {
                el.dataset.attack = String((+el.dataset.attack) + ra);
                updateAtk(el);
                showFloat(el, `🔥+${ra} ⚔️`, '#ff8a80');
                SFX.play('buff');
            }
            await sleep(150);
        }
    }
    async function runSideTurnStart(isP, s) {
        const cards = [];
        document.querySelectorAll('#game-board .card').forEach(el => {
            if (el.dataset.isPlayer === String(isP) && isAlive(el)) cards.push(el);
        });
        for (const el of cards) {
            if (s !== battleSeq || gameOver) return;
            await runPhase(el, 'onTurnStart', null, s);
        }
    }

    // ================= برد / باخت =================
    function scheduleWinCheck() {
        clearTimeout(winTimer);
        winTimer = setTimeout(checkWin, 1000);
    }
    function countAliveSide(isP) {
        let n = 0;
        for (let r = isP?3:0; r < (isP?6:3); r++)
            for (let c = 0; c < 3; c++) if (cardAt(r, c)) n++;
        return n;
    }
    function kingAlive(isP) {
        for (let r = isP?3:0; r < (isP?6:3); r++)
            for (let c = 0; c < 3; c++) {
                const el = cardAt(r, c);
                if (el && el.dataset.type === 'king') return true;
            }
        return false;
    }
    function checkWin() {
        if (gameOver) return;
        const eK = kingAlive(false), pK = kingAlive(true);
        const eN = countAliveSide(false), pN = countAliveSide(true);
        if (!eK || eN === 0) endGame(true);
        else if (!pK || pN === 0) endGame(false);
    }
    function endGame(playerWon) {
        if (gameOver) return;
        gameOver = true;
        stats.played++;
        if (playerWon) stats.wins++; else stats.losses++;
        saveLS('nabard_stats_v2', stats);
        SFX.play(playerWon ? 'victory' : 'defeat');
        logAdd(playerWon ? '🏆 شما برنده شدید!' : '💀 دشمن برنده شد', 'log-turn');
        setTimeout(() => {
            $('ov-title').textContent = playerWon ? '🏆 پیروزی!' : '💀 شکست!';
            $('ov-title').style.color = playerWon ? '#ffd54f' : '#ff5252';
            $('ov-sub').textContent = playerWon ? 'دشمنان شما نابود شدند!' : 'نیروهای شما نابود شدند!';
            $('ov-stats').innerHTML =
                `<div>🎯 تعداد نوبت‌ها: ${turnCount}</div>` +
                `<div>🥇 برد: ${stats.wins} | 🥉 باخت: ${stats.losses}</div>` +
                `<div>⚔️ کل آسیب تاریخچه: ${stats.damage}</div>`;
            $('overlay').style.display = 'flex';
        }, 700);
    }

    // ================= هوش مصنوعی دشمن =================
    async function enemyTurnLoop(s) {
        await sleep(600);
        while (!gameOver && s === battleSeq && currentTurn === 'enemy') {
            const atk = pickEnemyAttacker();
            if (!atk) break;
            const tgt = pickTargetFor(atk);
            if (!tgt) { atk.dataset.hasAttacked = 'true'; atk.classList.remove('can-attack'); continue; }
            await performAttack(atk, tgt);
            if (gameOver || s !== battleSeq) { busy = false; return; }
            await sleep(320);
        }
        if (s === battleSeq && !gameOver) await startPlayerTurn();
        else busy = false;
    }
    function pickEnemyAttacker() {
        const list = [];
        for (let r = 0; r < 3; r++) for (let c = 0; c < 3; c++) {
            const el = cardAt(r, c);
            if (el && el.dataset.hasAttacked !== 'true' && el.dataset.frozen !== '1' && el.dataset.type !== 'spell')
                list.push(el);
        }
        return list.length ? list[Math.floor(Math.random() * list.length)] : null;
    }
    function protectedBy(row, col) {
        if (row === 5) return cardAt(4, col);
        if (row === 4) return cardAt(3, col);
        return null;
    }
    function pickTargetFor(attacker) {
        const all = [];
        for (let r = 3; r < 6; r++) for (let c = 0; c < 3; c++) {
            const el = cardAt(r, c);
            if (el) all.push({ el, row: r, col: c });
        }
        if (!all.length) return null;
        const sniper = attacker.dataset.ignoreGuard === '1';
        let pool = sniper ? all : all.filter(o => {
            const g = protectedBy(o.row, o.col);
            return !(g && (g.dataset.type === 'defender' || g.dataset.type === 'both'));
        });
        if (!pool.length) pool = all;
        const nonStealth = pool.filter(o => o.el.dataset.stealth !== '1');
        if (nonStealth.length) pool = nonStealth;
        if (!sniper) {
            const taunts = pool.filter(o => o.el.dataset.taunt === '1');
            if (taunts.length) pool = taunts;
        }
        const score = o => {
            const t = o.el.dataset.type;
            let sc = t === 'king' ? 60 : t === 'mage' ? 45 : t === 'attacker' ? 35 : t === 'both' ? 25 : t === 'god' ? 30 : 15;
            if ((+attacker.dataset.attack) >= (+o.el.dataset.health)) sc += 30;
            sc += ((+o.el.dataset.maxHealth) - (+o.el.dataset.health)) * 0.5;
            return sc;
        };
        let best = null, bs = -1;
        pool.forEach(o => { const sc = score(o); if (sc > bs) { bs = sc; best = o; } });
        return best.el;
    }

                const slot = document.createElement('div');

    function onEditorSlotClick(r, c) {
        const cur = editorLayout[r][c];
        if (cur && cur.kind !== 'default') {
            editorLayout[r][c] = defaultEntry(r, c);
            toast('کارت حذف شد');
        } else {
            editorLayout[r][c] = placingDef
                ? { kind: 'custom', data: deepCopy(placingDef) }
                : { kind: 'draft' };
            SFX.play('select');
        }
        layoutDirty = true;
        renderEditorBoard();
    }
    function renderEditorBoard() {
        document.querySelectorAll('#editor-board .row').forEach(rowEl => {
            const r = +rowEl.dataset.erow;
            for (let c = 0; c < 3; c++) {
                const slot = rowEl.children[c];
                if (!slot) continue;
                slot.innerHTML = '';
                const e = editorLayout[r][c];
                if (!e) continue;
                let el;
                if (e.kind === 'default') {
                    const d = CARD_DATA[e.id];
                    el = makeCardEl({...d, id: e.id}, r >= 3, false);
                } else if (e.kind === 'draft') {
                    el = makeCardEl(deepCopy(editorCard), r >= 3, true);
                } else {
                    el = makeCardEl(deepCopy(e.data), r >= 3, true);
                }
                el.classList.add('static');
                slot.appendChild(el);
            }
        });
    }

    function handleImageUpload(e) {
        const file = e.target.files[0];
        if (!file) return;
        const reader = new FileReader();
        reader.onload = ev => {
            const img = new Image();
            img.onload = () => {
                const MAX = 180;
                const scale = Math.min(1, MAX / Math.max(img.width, img.height));
                const cv = document.createElement('canvas');
                cv.width = Math.round(img.width * scale);
                cv.height = Math.round(img.height * scale);
                cv.getContext('2d').drawImage(img, 0, 0, cv.width, cv.height);
                editorCard.image = cv.toDataURL('image/jpeg', 0.82);
                layoutDirty = true;
                renderPreview(); renderEditorBoard();
                toast('✅ عکس اضافه شد');
            };
            img.src = ev.target.result;
        };
        reader.readAsDataURL(file);
    }

            chip.innerHTML = `${def.icon} ${def.label}`;

                    editorCard.passives[def.key + '_v'] = def.def;

    const enc = o => btoa(unescape(encodeURIComponent(JSON.stringify(o))));
    const dec = s => JSON.parse(decodeURIComponent(escape(atob(s.trim()))));
    function normalizeSounds(s) {
        const ok = (v) => (typeof v === 'string' && v.startsWith('data:audio')) ? v : null;
        return { start: ok(s && s.start), attack: ok(s && s.attack), death: ok(s && s.death) };
    }
    function normalizePassives(p) {
        p = p || {};
        const num = (v, max = 99) => clamp(+v || 0, 0, max);
        return {
            doubleStrike: !!p.doubleStrike,
            cleave: !!p.cleave,
            crit: num(p.crit),
            resurrect: num(p.resurrect, 100),
            lastStand: !!p.lastStand,
            regen: num(p.regen),
            berserk: num(p.berserk),
            rage: num(p.rage),
            pierce: !!p.pierce,
            ignoreGuard: !!p.ignoreGuard,
            stealth: !!p.stealth
        };
    }
    function sanitizeAb(a) {
        const out = { onStart: [], onTurnStart: [], onAttack: [], onDefend: [], onKill: [], onDeath: [] };
        if (!a || typeof a !== 'object') return out;
        Object.keys(out).forEach(ph => {
            if (!Array.isArray(a[ph])) return;
            a[ph].forEach(it => {
                if (!it || !KNOWN_ACTIONS.includes(it.action)) return;
                const positions = (Array.isArray(it.positions) ? it.positions : [])
                    .filter(p => Array.isArray(p) && p[0] >= 0 && p[0] < 6 && p[1] >= 0 && p[1] < 3)
                    .map(p => [+p[0], +p[1]]);
                out[ph].push({
                    action: it.action, label: String(it.label || it.action), icon: String(it.icon || '⚡'),
                    amount: clamp(+it.amount || 0, 0, 99),
                    targetSide: it.targetSide === 'ally' ? 'ally' : 'enemy',
                    positions
                });
            });
        });
        return out;
    }
    function normalizeDef(d, keepId) {
        return {
            id: keepId ? d.id : ('cc_' + Date.now() + Math.floor(Math.random() * 999)),
            name: String(d.name || 'کارت').slice(0, 20),
            type: TYPE_LABELS[d.type] ? d.type : 'attacker',
            tier: ['normal','medium','legendary','god'].includes(d.tier) ? d.tier : 'normal',
            attack: clamp(+d.attack || 0, 0, 99),
            health: clamp(+d.health || 1, 1, 99),
            icon: String(d.icon || '❓').slice(0, 4),
            image: (typeof d.image === 'string' && d.image.startsWith('data:image')) ? d.image : null,
            borderColor: /^#[0-9a-fA-F]{6}$/.test(d.borderColor || '') ? d.borderColor : '#ffb300',
            sounds: normalizeSounds(d.sounds),
            passives: normalizePassives(d.passives),
            abilities: sanitizeAb(d.abilities)
        };
    }
    function renderLibrary() {
        const list = $('library-list');
        list.innerHTML = '';
        if (!customCards.length) {
            list.innerHTML = '<div class="lib-empty">هنوز کارتی ذخیره نشده است</div>';

            const pvN = Object.values(cc.passives || {}).filter(v => v === true || (+v > 0)).length;

            item.innerHTML = `

                <div class="lib-name">${esc(cc.name || 'بدون نام')}</div>
                <div class="lib-stats">⚔️${cc.attack} ❤️${cc.health} ${abN ? '⚡' + abN : ''} ${pvN ? '🔥' + pvN : ''} ${sdN ? '🔊' + sdN : ''}</div>
                <button class="lib-load">📥</button><button class="lib-place">📍</button>
                <button class="lib-dup">📋</button><button class="lib-exp">📤</button>
                <button class="lib-del">🗑</button>`;
            item.querySelector('.lib-load').addEventListener('click', () => {

                $('place-banner').style.display = 'block';
                $('place-name').textContent = cc.name || 'کارت';

                saveLS('nabard_custom_cards_v2', customCards);
                renderLibrary();
            });

        });

    function showCardInfo(el) {
        let ab = {}; try { ab = JSON.parse(el.dataset.abilities || '{}'); } catch(e){}
        const rows = [];
        Object.keys(PHASE_LABELS).forEach(ph => {
            (ab[ph] || []).forEach(a => {
                const meta = ACTION_META[a.action] || {};
                const targetTxt = meta.auto ? 'خودکار 🎯' : (a.targetSide === 'ally' ? 'خودی' : 'دشمن');
                rows.push(`<div class="info-ab"><b>${PHASE_LABELS[ph]}:</b> ${a.icon} ${esc(a.label)} — ${meta.unit || 'مقدار'}: ${a.amount} — هدف: ${targetTxt}</div>`);
            });
        });
        const pv = [];
        PASSIVE_DEFS.forEach(d => {
            const v = el.dataset[d.key];
            const active = d.hasAmount ? (+v > 0) : (v === '1');
            if (active) pv.push(`${d.icon} <b>${d.label}</b>${d.hasAmount ? ' (' + v + (d.unit || '') + ')' : ''} — ${d.desc}`);
        });
        const st = [];
        if (+el.dataset.shield > 0) st.push('🛡️ سپر ' + el.dataset.shield);
        if (+el.dataset.thorns > 0) st.push('🌵 خار ' + el.dataset.thorns);
        if (+el.dataset.poison > 0) st.push('☠️ زهر ' + el.dataset.poison);
        if (+el.dataset.burn > 0) st.push('🔥 آتش ' + el.dataset.burn);
        if (+el.dataset.dodge > 0) st.push('💨 جاخالی ' + el.dataset.dodge + '%');
        if (el.dataset.taunt === '1') st.push('🎯 تحریک‌شده');
        if (el.dataset.silenced === '1') st.push('🔇 ساکت');
        if (el.dataset.frozen === '1') st.push('❄️ یخ‌زده');
        if (el.dataset.soundStart || el.dataset.soundAttack || el.dataset.soundDeath) st.push('🔊 صدای شخصی');
        $('ci-body').innerHTML = `
            <div class="ci-head">${esc(el.dataset.name)}</div>
            <div class="ci-stats">${TYPE_LABELS[el.dataset.type] || ''} | ⚔️ ${el.dataset.attack} | ❤️ ${el.dataset.health}/${el.dataset.maxHealth}
            ${st.length ? '<br>' + st.join(' | ') : ''}</div>
            ${pv.length ? '<div class="info-ab"><b>🔥 پسیوها:</b><br>' + pv.join('<br>') + '</div>' : ''}
            ${rows.length ? rows.join('') : (pv.length ? '' : '<div class="info-ab">این کارت قابلیت ویژه‌ای ندارد</div>')}`;
        $('card-info-overlay').style.display = 'flex';
    }
