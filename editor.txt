<script>
(function() {
  'use strict';

  const preview   = document.getElementById('preview');
  const startBtn  = document.getElementById('startBtn');
  const btnLabel  = document.getElementById('btnLabel');
  const bgInput   = document.getElementById('bgInput');
  const btnInput  = document.getElementById('btnInput');
  const btnText   = document.getElementById('btnText');
  const btnW      = document.getElementById('btnW');
  const btnH      = document.getElementById('btnH');
  const btnX      = document.getElementById('btnX');
  const btnY      = document.getElementById('btnY');
  const wVal = document.getElementById('wVal');
  const hVal = document.getElementById('hVal');
  const xVal = document.getElementById('xVal');
  const yVal = document.getElementById('yVal');
  const panel  = document.getElementById('panel');
  const togglePanelBtn = document.getElementById('togglePanel');
  const closePanelBtn  = document.getElementById('closePanel');
  const editBgBtn = document.getElementById('editBgBtn');
  const editBtnBtn = document.getElementById('editBtnBtn');

  let bgData = '';
  let btnData = '';
  let dragging = false;

  const isMobile = () => window.matchMedia('(max-width: 900px)').matches;

  function updateCloseBtn() {
    closePanelBtn.style.display = isMobile() ? 'block' : 'none';
  }
  updateCloseBtn();
  window.addEventListener('resize', updateCloseBtn);

  togglePanelBtn.addEventListener('click', () => {
    panel.classList.toggle('open');
    if (panel.classList.contains('open')) {
      setTimeout(() => panel.scrollIntoView({ behavior: 'smooth', block: 'end' }), 100);
    }
  });
  closePanelBtn.addEventListener('click', () => panel.classList.remove('open'));

  // ====== آپلود تصاویر ======
  bgInput.addEventListener('change', e => {
    const f = e.target.files[0];
    if (!f) return;
    const r = new FileReader();
    r.onload = ev => {
      bgData = ev.target.result;
      preview.style.backgroundImage = `url(${bgData})`;
      editBgBtn.style.display = 'block';
    };
    r.readAsDataURL(f);
  });

  btnInput.addEventListener('change', e => {
    const f = e.target.files[0];
    if (!f) return;
    const r = new FileReader();
    r.onload = ev => {
      btnData = ev.target.result;
      startBtn.style.background = `url(${btnData}) center/100% 100% no-repeat`;
      startBtn.classList.remove('btnDefault');
      btnLabel.style.display = 'none';
      editBtnBtn.style.display = 'block';
    };
    r.readAsDataURL(f);
  });

  btnText.addEventListener('input', () => {
    if (!btnData) btnLabel.textContent = btnText.value || 'شروع';
  });

  // ====== ابعاد و موقعیت دکمه ======
  function updateSize() {
    startBtn.style.width = btnW.value + 'px';
    startBtn.style.height = btnH.value + 'px';
    wVal.textContent = btnW.value + 'px';
    hVal.textContent = btnH.value + 'px';
  }
  btnW.addEventListener('input', updateSize);
  btnH.addEventListener('input', updateSize);

  function updatePos() {
    startBtn.style.left = btnX.value + '%';
    startBtn.style.top  = btnY.value + '%';
    xVal.textContent = btnX.value + '%';
    yVal.textContent = btnY.value + '%';
  }
  btnX.addEventListener('input', updatePos);
  btnY.addEventListener('input', updatePos);

  // ====== درگ دکمه ======
  function getPoint(e) {
    const rect = preview.getBoundingClientRect();
    let clientX, clientY;
    if (e.touches && e.touches.length) {
      clientX = e.touches[0].clientX; clientY = e.touches[0].clientY;
    } else if (e.changedTouches && e.changedTouches.length) {
      clientX = e.changedTouches[0].clientX; clientY = e.changedTouches[0].clientY;
    } else {
      clientX = e.clientX; clientY = e.clientY;
    }
    let x = clientX - rect.left;
    let y = clientY - rect.top;
    x = Math.max(0, Math.min(rect.width,  x));
    y = Math.max(0, Math.min(rect.height, y));
    return { xPct: (x / rect.width) * 100, yPct: (y / rect.height) * 100 };
  }
  function applyDragPoint(e) {
    const p = getPoint(e);
    btnX.value = Math.round(p.xPct);
    btnY.value = Math.round(p.yPct);
    updatePos();
  }
  function startDrag(e) {
    dragging = true;
    e.preventDefault(); e.stopPropagation();
  }
  function endDrag() { dragging = false; }

  startBtn.addEventListener('mousedown', startDrag);
  document.addEventListener('mousemove', e => { if (dragging) applyDragPoint(e); });
  document.addEventListener('mouseup', endDrag);

  startBtn.addEventListener('touchstart', startDrag, { passive: false });
  document.addEventListener('touchmove', e => {
    if (!dragging) return;
    e.preventDefault();
    applyDragPoint(e);
  }, { passive: false });
  document.addEventListener('touchend', endDrag);
  document.addEventListener('touchcancel', endDrag);

  preview.addEventListener('gesturestart', e => e.preventDefault());

  // ==================================================================
  // ==================== مودال ویرایش تصویر ==========================
  // ==================================================================
  const editModal   = document.getElementById('editModal');
  const editTitle   = document.getElementById('editTitle');
  const editCanvas  = document.getElementById('editCanvas');
  const ctx         = editCanvas.getContext('2d');
  const canvasWrap  = document.getElementById('canvasWrap');
  const cropBox     = document.getElementById('cropBox');
  const removeColor = document.getElementById('removeColor');
  const tolerance   = document.getElementById('tolerance');
  const tolVal      = document.getElementById('tolVal');
  const feather     = document.getElementById('feather');
  const featherVal  = document.getElementById('featherVal');
  const pickColorBtn= document.getElementById('pickColorBtn');
  const applyRemove = document.getElementById('applyRemove');
  const resetEdit   = document.getElementById('resetEdit');
  const applyCrop   = document.getElementById('applyCrop');
  const cropReset   = document.getElementById('cropReset');
  const modalApply  = document.getElementById('modalApply');
  const modalCancel = document.getElementById('modalCancel');

  // state مودال
  let editingTarget = null;      // 'bg' | 'btn'
  let originalImage = null;      // Image اصلی (برای بازگردانی)
  let currentCanvas = null;      // canvas کاری
  let baseImageData = null;      // داده‌ی اولیه برای بازگردانی سریع
  let pickingColor  = false;     // حالت انتخاب رنگ از تصویر
  let cropStart     = null;      // شروع کشیدن ناحیه برش
  let cropRect      = null;      // { x, y, w, h } نسبت به canvas
  let displayScale  = 1;         // نسبت canvas به نمایش

  tolerance.addEventListener('input', () => tolVal.textContent = tolerance.value);
  feather.addEventListener('input', () => featherVal.textContent = feather.value);

  // تب‌ها
  document.querySelectorAll('#editTools .tabs button').forEach(btn => {
    btn.addEventListener('click', () => {
      const tab = btn.dataset.tab;
      document.querySelectorAll('#editTools .tabs button').forEach(b => b.classList.remove('active'));
      document.querySelectorAll('.tab-content').forEach(c => c.classList.remove('active'));
      btn.classList.add('active');
      document.querySelector(`.tab-content[data-content="${tab}"]`).classList.add('active');

      if (tab === 'crop') {
        canvasWrap.style.cursor = 'crosshair';
      } else {
        canvasWrap.style.cursor = 'default';
        cropBox.style.display = 'none';
        cropRect = null;
      }
    });
  });

  // باز کردن مودال
  function openEditor(target) {
    editingTarget = target;
    editTitle.textContent = target === 'bg' ? '✂️ ویرایش پس‌زمینه' : '✂️ ویرایش تصویر دکمه';
    const src = target === 'bg' ? bgData : btnData;
    if (!src) return;

    const img = new Image();
    img.onload = () => {
      originalImage = img;
      // canvas با اندازه‌ی تصویر اصلی
      editCanvas.width  = img.naturalWidth;
      editCanvas.height = img.naturalHeight;
      ctx.clearRect(0, 0, editCanvas.width, editCanvas.height);
      ctx.drawImage(img, 0, 0);
      baseImageData = ctx.getImageData(0, 0, editCanvas.width, editCanvas.height);

      // نمایش canvas
      fitCanvasDisplay();
      cropBox.style.display = 'none';
      cropRect = null;

      // ریست UI
      resetToolStates();

      editModal.classList.add('open');
    };
    img.src = src;
  }

  // تنظیم اندازه‌ی نمایش canvas
  function fitCanvasDisplay() {
    const wrapRect = canvasWrap.getBoundingClientRect();

  function rgbToHex(r, g, b) {
    return '#' + [r, g, b].map(v => v.toString(16).padStart(2, '0')).join('');
  }
  function hexToRgb(hex) {
    const m = hex.match(/^#?([a-f\d]{2})([a-f\d]{2})([a-f\d]{2})$/i);
    return m ? { r: parseInt(m[1],16), g: parseInt(m[2],16), b: parseInt(m[3],16) } : { r:0,g:0,b:0 };
  }

  function canvasCoordsFromEvent(e) {
    const rect = editCanvas.getBoundingClientRect();
    let clientX, clientY;
    if (e.touches && e.touches.length) {
      clientX = e.touches[0].clientX; clientY = e.touches[0].clientY;
    } else if (e.changedTouches && e.changedTouches.length) {
      clientX = e.changedTouches[0].clientX; clientY = e.changedTouches[0].clientY;
    } else {
      clientX = e.clientX; clientY = e.clientY;
    }
    const x = (clientX - rect.left) * (editCanvas.width  / rect.width);
    const y = (clientY - rect.top)  * (editCanvas.height / rect.height);
    return { x: Math.max(0, Math.min(editCanvas.width  - 1, Math.round(x))),
             y: Math.max(0, Math.min(editCanvas.height - 1, Math.round(y))) };
  }

  // ====== حذف رنگ (Flood-fill style flood-fill یا global) ======
  // از روش «حذف تمام پیکسل‌های مشابه در کل تصویر» با آستانه استفاده می‌کنیم
  function removeColorFromImage(targetRgb, tol, featherAmount) {
    const imgData = ctx.getImageData(0, 0, editCanvas.width, editCanvas.height);
    const data = imgData.data;
    const len = data.length;
    const { r: tr, g: tg, b: tb } = targetRgb;
    const tolSq = tol * tol * 3; // برای سرعت

    for (let i = 0; i < len; i += 4) {
      const r = data[i], g = data[i+1], b = data[i+2];
      const dr = r - tr, dg = g - tg, db = b - tb;
      const distSq = dr*dr + dg*dg + db*db;

      if (distSq <= tolSq) {
        // شفاف کامل
        data[i+3] = 0;
      } else if (featherAmount > 0) {
        // نرمی لبه‌ها: پیکسل‌های نزدیک به آستانه، نیمه شفاف
        const featherThreshold = (tol + featherAmount) * (tol + featherAmount) * 3;
        if (distSq <= featherThreshold) {
          const t = Math.sqrt(distSq);
          const a = 1 - (t - tol) / featherAmount;
          data[i+3] = Math.max(0, Math.min(255, data[i+3] * a));
        }
      }
    }
    ctx.putImageData(imgData, 0, 0);
  }

  function onCanvasDown(e) {
    // اگر حالت انتخاب رنگ فعال است
    if (pickingColor) {
      const p = canvasCoordsFromEvent(e);
      const px = ctx.getImageData(p.x, p.y, 1, 1).data;

      canvasWrap.style.cursor = 'default';
      e.preventDefault();
      return;
    }

    // اگر تب برش فعال است
    const cropTabActive = document.querySelector('#editTools .tabs button.active')?.dataset.tab === 'crop';
    if (!cropTabActive) return;

    const p = canvasCoordsFromEvent(e);
    cropStart = p;
    cropRect = { x: p.x, y: p.y, w: 0, h: 0 };
    redrawCropBox();
    canvasPointerActive = true;
    e.preventDefault();
  }

  function onCanvasMove(e) {
    if (!canvasPointerActive || !cropStart) return;
    const p = canvasCoordsFromEvent(e);
    const x = Math.min(cropStart.x, p.x);
    const y = Math.min(cropStart.y, p.y);
    const w = Math.abs(p.x - cropStart.x);
    const h = Math.abs(p.y - cropStart.y);
    cropRect = { x, y, w, h };
    redrawCropBox();
    e.preventDefault();
  }

  function onCanvasUp() {
    canvasPointerActive = false;
  }

  function redrawCropBox() {
    if (!cropRect || cropRect.w < 2 || cropRect.h < 2) {
      cropBox.style.display = 'none';
      return;
    }
    const canvasRect = editCanvas.getBoundingClientRect();
    const wrapRect = canvasWrap.getBoundingClientRect();
    const offsetX = canvasRect.left - wrapRect.left;
    const offsetY = canvasRect.top  - wrapRect.top;
    const scaleX = canvasRect.width  / editCanvas.width;
    const scaleY = canvasRect.height / editCanvas.height;

    cropBox.style.display = 'block';
    cropBox.style.left   = (offsetX + cropRect.x * scaleX) + 'px';
    cropBox.style.top    = (offsetY + cropRect.y * scaleY) + 'px';
    cropBox.style.width  = (cropRect.w * scaleX) + 'px';
    cropBox.style.height = (cropRect.h * scaleY) + 'px';
  }
